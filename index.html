    // 自動容錯載入圖片（同時支援 .jpg / .jpeg / 大寫副檔名）

    function setSmartImage(imgEl, baseName, preferredExts) {

      let idx = 0;

      function tryNext() {

        if (idx >= preferredExts.length) return;

        imgEl.src = baseName + "." + preferredExts[idx++];

      }

      imgEl.onerror = tryNext;

      tryNext();

    }


    // 解析可用的圖片 URL（供拼圖背景使用）

    function resolveImageUrl(baseName, exts, callback) {

      let idx = 0;

      const tester = new Image();

      function tryNext() {

        if (idx >= exts.length) {

          callback(baseName + "." + exts[0]);

          return;

        }

        const candidate = baseName + "." + exts[idx++];

        tester.onload = () => callback(candidate);

        tester.onerror = tryNext;

        tester.src = candidate;

      }

      tryNext();

    }

    /* ================= 任務一：回憶對對樂 ================= */

    const pairSources = [

      { base: "IMG_01", exts: ["jpeg", "jpg", "JPEG", "JPG"] },

      { base: "IMG_02", exts: ["jpg", "jpeg", "JPG", "JPEG"] },

      { base: "IMG_03", exts: ["jpeg", "jpg", "JPEG", "JPG"] },

      { base: "IMG_04", exts: ["jpg", "jpeg", "JPG", "JPEG"] },

      { base: "IMG_05", exts: ["jpg", "jpeg", "JPG", "JPEG"] },

      { base: "IMG_06", exts: ["jpeg", "jpg", "JPEG", "JPG"] }

    ];


    let flippedCards = [];

    let matchedPairs = 0;

    let lockBoard = false;


    function initMemoryGame() {

      const grid = document.getElementById("memory-grid");

      grid.innerHTML = "";

      const deck = [...pairSources, ...pairSources]

        .map((item, idx) => ({ ...item, uid: idx, sort: Math.random() }))

        .sort((a, b) => a.sort - b.sort);


      deck.forEach(item => {

        const card = document.createElement("div");

        card.className = "mem-card";

        card.dataset.base = item.base;


        const inner = document.createElement("div");

        inner.className = "mem-inner";


        const front = document.createElement("div");

        front.className = "mem-face mem-front";

        front.innerHTML = "♥<span>LOVE</span>";


        const back = document.createElement("div");

        back.className = "mem-face mem-back";

        const img = document.createElement("img");

        img.alt = item.base;

        setSmartImage(img, item.base, item.exts);

        back.appendChild(img);


        inner.appendChild(front);

        inner.appendChild(back);

        card.appendChild(inner);


        card.addEventListener("click", () => handleFlip(card));

        grid.appendChild(card);

      });

    }


    function handleFlip(card) {

      if (lockBoard || card.classList.contains("flipped") || card.classList.contains("matched")) return;

      card.classList.add("flipped");

      flippedCards.push(card);


      if (flippedCards.length === 2) {

        lockBoard = true;

        const [c1, c2] = flippedCards;

        if (c1.dataset.base === c2.dataset.base) {

          c1.classList.add("matched");

          c2.classList.add("matched");

          flippedCards = [];

          lockBoard = false;

          matchedPairs++;

          document.getElementById("match-count").textContent = matchedPairs;


          if (matchedPairs === 6) {

            setTimeout(() => {

              alert("🎉 太棒了！6 組回憶照片全部配對成功，任務一已入帳！");

              markTaskDone(1);

              openView("view-home");

            }, 450);

          }

        } else {

          setTimeout(() => {

            c1.classList.remove("flipped");

            c2.classList.remove("flipped");

            flippedCards = [];

            lockBoard = false;

          }, 750);

        }

      }

    }

    /* ================= 任務三：照片拼拼圖 ================= */

    let puzzleUrl = "IMG_m3.jpg";

    let tileOrder = [0, 1, 2, 3, 4, 5, 6, 7, 8];

    let selectedSlot = null;


    function initPuzzle() {

      selectedSlot = null;

      // 洗牌且確保不是已經完成的順序

      do {

        tileOrder = [0, 1, 2, 3, 4, 5, 6, 7, 8].sort(() => Math.random() - 0.5);

      } while (tileOrder.every((val, idx) => val === idx));

      renderPuzzle();

    }


    function renderPuzzle() {

      const grid = document.getElementById("puzzle-grid");

      grid.innerHTML = "";


      tileOrder.forEach((pieceIdx, slotIdx) => {

        const tile = document.createElement("div");

        tile.className = "puzzle-tile" + (selectedSlot === slotIdx ? " selected" : "");

        const col = pieceIdx % 3;

        const row = Math.floor(pieceIdx / 3);

        tile.style.backgroundImage = `url('${puzzleUrl}')`;

        tile.style.backgroundSize = "300% 300%";

        tile.style.backgroundPosition = `${col * 50}% ${row * 50}%`;


        tile.addEventListener("click", () => handleTileClick(slotIdx));

        grid.appendChild(tile);

      });

    }


    function handleTileClick(slotIdx) {

      if (selectedSlot === null) {

        selectedSlot = slotIdx;

        renderPuzzle();

        return;

      }

      if (selectedSlot === slotIdx) {

        selectedSlot = null;

        renderPuzzle();

        return;

      }


      // 交換兩塊拼圖

      const temp = tileOrder[selectedSlot];

      tileOrder[selectedSlot] = tileOrder[slotIdx];

      tileOrder[slotIdx] = temp;

      selectedSlot = null;

      renderPuzzle();


      // 檢查是否拼圖完成

      if (tileOrder.every((val, idx) => val === idx)) {

        setTimeout(() => {

          alert("🧩 拼圖完成！最可愛的女友出現啦，任務三已入帳！");

          markTaskDone(3);

          openView("view-home");

        }, 350);

      }

    }

    }
